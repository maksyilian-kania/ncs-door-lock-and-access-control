# Vendors this model's own edge-impulse-sdk/ copy directly (wholesale, not
# cherry-picked) instead of depending on the CONFIG_EDGE_IMPULSE_SDK
# west-module/FetchContent path used by e.g. edge-ai's hello_ei sample.
# Reasons, per Camera driver plan v2.md section 8:
#  - edge-impulse-sdk/cmake/zephyr/CMakeLists.txt does a recursive glob over
#    the whole tree (plus a hand-picked set of CMSIS/DSP + CMSIS/NN
#    subfolders) and relies on internal #ifdefs, keyed off
#    EI_CLASSIFIER_INFERENCING_ENGINE from model_metadata.h, to compile out
#    unused backends/platforms as near-empty translation units. Cherry-
#    picking risks breaking that glob/those relative includes for no real
#    flash/RAM benefit.
#  - The model-parameters/ and nordic-axon-model/ folder *names* are
#    load-bearing: edge-impulse-sdk/classifier/inferencing_engines/nordic_axon.h
#    does a bare #include "model-parameters/model_metadata.h" and resolves
#    the Axon model header via NRF_AXON_MODEL_NAME through the include
#    search path, not a relative path.
#
# NOTE (not build-verified, no west/toolchain available in this workspace):
# when CONFIG_EDGE_IMPULSE_SDK is *not* defined (our case here),
# edge-impulse-sdk/cmake/zephyr/CMakeLists.txt's else() branch adds its
# sources/includes directly against the Zephyr `app` target
# (target_sources(app ...)/target_include_directories(app ...)), not via
# zephyr_library(). That requires the `app` target to already exist at the
# point add_subdirectory() below runs. This has not been exercised with a
# real `west build` - check it first if the build fails to find `app`.
add_subdirectory(edge-impulse-sdk/cmake/zephyr)

zephyr_include_directories(.)
zephyr_include_directories(nordic-axon-model)

# Matches the compile definitions edge-ai's hello_ei sample uses for the same
# vendored-SDK-called-from-C++ setup.
zephyr_compile_definitions(
  NRF_AXON_MODEL_NAME=${CONFIG_NRF_AXON_MODEL_NAME}
  EI_C_LINKAGE=1
  EIDSP_SIGNAL_C_FN_POINTER=1
)

# Suppress warnings generated by the vendored Edge Impulse SDK (same set
# hello_ei suppresses).
zephyr_compile_options(
  -Wno-double-promotion
  -Wno-unused
  -Wno-sign-compare
  -Wno-maybe-uninitialized
)
